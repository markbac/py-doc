% ---------------------------------------------------------------------------
% Fonts (XeLaTeX / fontspec) — Carlito/Caladea are metric-compatible with
% Calibri/Cambria, matching the look of the Template.docx output so the
% XeLaTeX and LibreOffice-converted PDFs stay visually consistent.
% Where they are not installed (a TeX outside the tool container), the fonts
% they copy come next (Cambria, Calibri, Consolas on Windows), then Latin Modern.
% Their fi and fl ligatures copy out of a PDF as nothing, so they are off.
% ---------------------------------------------------------------------------
\usepackage{fontspec}
\IfFontExistsTF{Caladea}{\setmainfont{Caladea}}{%
  \IfFontExistsTF{Cambria}{\setmainfont{Cambria}[Ligatures=NoCommon]}{\setmainfont{Latin Modern Roman}}}
\IfFontExistsTF{Carlito}{\setsansfont{Carlito}}{%
  \IfFontExistsTF{Calibri}{\setsansfont{Calibri}[Ligatures=NoCommon]}{\setsansfont{Latin Modern Sans}}}
\IfFontExistsTF{DejaVu Sans Mono}{\setmonofont{DejaVu Sans Mono}[Scale=0.88]}{%
  \IfFontExistsTF{Consolas}{\setmonofont{Consolas}[Scale=0.88]}{\setmonofont{Latin Modern Mono}}}
% Caladea and Carlito have no Greek letters and no subscript or superscript digits, so "4.7 kΩ" or "CO₂" would
% lose them. Each of those is drawn by DejaVu Serif where the font in use has no glyph for it.
\IfFontExistsTF{Caladea}{\IfFontExistsTF{DejaVu Serif}{%
  \usepackage{newunicodechar}
  \newfontfamily\pydocsymbols{DejaVu Serif}[Scale=MatchLowercase]
  \def\pydocFallback#1{\expandafter\newunicodechar\expandafter{\Uchar#1}%
    {\iffontchar\font#1 \char#1 \else{\pydocsymbols\char#1}\fi}}
  \def\pydocRange#1#2{\count255=#1 \loop\ifnum\count255<#2 \edef\pydocN{\the\count255}%
    \expandafter\pydocFallback\expandafter{\pydocN}\advance\count255 by 1 \repeat}
  \pydocRange{913}{938}\pydocRange{945}{970}% Greek capitals and small letters
  \pydocRange{8304}{8335}% superscript and subscript digits and signs
}{}}{}

% sentinel to suppress numbering. The caption package's longtable patch
% (ltcaption.sty, auto-loaded above) looks up \LTcaptype as a real counter

% Defining a real (unused) counter called "none" satisfies that lookup.

% Clamp every \includegraphics call to fit the page -- shrinks oversized
% images, never stretches small ones. Hooking \includegraphics itself
% (rather than only \pandocbounded) makes this work regardless of Pandoc
% version: older Pandoc emits bare \includegraphics with no size wrapper
% at all; newer Pandoc wraps it in \pandocbounded. Centering is left to
% the enclosing figure environment (which already applies \centering) --
% forcing it here too would wrongly break inline (non-figure) images
% out of their surrounding text.
\let\oldincludegraphics\includegraphics
\renewcommand{\includegraphics}[2][]{%
  \adjustbox{max width=\linewidth,max height=0.8\textheight,keepaspectratio}{\oldincludegraphics[#1]{#2}}%
}
